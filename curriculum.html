// Cosine similarity implementation
function cosineSimilarity(a: number[], b: number[]): number {
  const dotProduct = a.reduce((sum, val, i) => sum + val * b[i], 0);
  const magnitudeA = Math.sqrt(a.reduce((sum, val) => sum + val * val, 0));
  const magnitudeB = Math.sqrt(b.reduce((sum, val) => sum + val * val, 0));
  return dotProduct / (magnitudeA * magnitudeB);
}

Theory Reciprocal Rank Fusion (RRF)

Learn how to combine results from multiple retrieval methods without normalizing incompatible scores.

// RRF Algorithm
function reciprocalRankFusion(
  rankings: Map<string, number>[],  // docId -> rank (1-indexed)
  k: number = 60  // constant, typically 60
): Map<string, number> {
  const scores = new Map<string, number>();
  
  for (const ranking of rankings) {
    for (const [docId, rank] of ranking) {
      const current = scores.get(docId) || 0;
      scores.set(docId, current + 1 / (k + rank));
    }
  }
  
  return scores;  // Higher = better
}

// Why RRF works:
// - Rank-based (ignores raw scores)
// - No normalization needed
// - Documents ranked highly by multiple methods bubble up
💡 Key Insight

RRF is "embarrassingly simple" but remarkably effective. It outperforms learned fusion methods in many cases because it doesn't overfit to specific score distributions.

Hands-On Token-Aware Result Filtering

Ensure retrieved chunks fit within LLM context limits.

import { encode } from 'gpt-tokenizer';

function filterToTokenBudget(
  chunks: Chunk[],
  maxTokens: number
): Chunk[] {
  let totalTokens = 0;
  const selected: Chunk[] = [];
  
  for (const chunk of chunks) {
    const chunkTokens = encode(chunk.content).length;
    if (totalTokens + chunkTokens > maxTokens) break;
    selected.push(chunk);
    totalTokens += chunkTokens;
  }
  
  return selected;
}

Hands-On Entity Resolution

Handle the same entity mentioned different ways: "Randy", "Randy Wilson", "R. Wilson".

// Entity resolution strategies
async function resolveEntity(
  newEntity: Entity,
  existingEntities: Entity[]
): Promise<Entity> {
  // 1. Exact name match
  const exactMatch = existingEntities.find(
    e => e.name.toLowerCase() === newEntity.name.toLowerCase()
  );
  if (exactMatch) return exactMatch;
  
  // 2. Fuzzy name match (Jaro-Winkler)
  const fuzzyMatches = existingEntities.filter(
    e => jaroWinkler(e.name, newEntity.name) > 0.85
  );
  
  // 3. LLM-based resolution for ambiguous cases
  if (fuzzyMatches.length > 0) {
    const resolution = await llm.resolve(newEntity, fuzzyMatches);
    if (resolution.isSameEntity) {
      return mergeEntities(resolution.matchedEntity, newEntity);
    }
  }
  
  // 4. Create new entity
  return newEntity;
}

Hands-On Dynamic Tool Management

Implement tool activation/deactivation to manage context window usage. Build a registry with semantic search for tool discovery.

💡 Key Insight

Tool schemas consume ~100-150 tokens each. With 40 tools, that's 4,000-6,000 tokens per API call—before any conversation history! The solution: only send schemas for tools the agent actually needs right now.

// Tool Registry with Semantic Search
interface ToolMetadata {
  name: string;
  description: string;
  keywords: string[];
  embedding?: number[];  // For semantic search
  active: boolean;
}

class ToolRegistry {
  private tools = new Map<string, ToolMetadata>();
  
  register(name: string, tool: any, metadata: Partial<ToolMetadata>) {
    this.tools.set(name, {
      name,
      description: tool.description || '',
      keywords: metadata.keywords || [],
      active: false,
    });
  }
  
  // Generate embeddings for semantic search
  async generateEmbeddings() {
    for (const [name, meta] of this.tools) {
      const searchText = `${name} ${meta.description} ${meta.keywords.join(' ')}`;
      meta.embedding = await embed(searchText);
    }
  }
  
  // Semantic search for relevant tools
  async searchSemantic(query: string, topK = 5): Promise<ToolMetadata[]> {
    const queryEmbedding = await embed(query);
    
    return [...this.tools.values()]
      .map(tool => ({
        tool,
        score: cosineSimilarity(queryEmbedding, tool.embedding!),
      }))
      .sort((a, b) => b.score - a.score)
      .slice(0, topK)
      .map(r => r.tool);
  }
  
  activate(name: string) { this.tools.get(name)!.active = true; }
  deactivate(name: string) { this.tools.get(name)!.active = false; }
  getActive(): string[] { 
    return [...this.tools.values()].filter(t => t.active).map(t => t.name);
  }
}

}

Deep Dive Tool Instrumentation Wrapper

Create a higher-order function that wraps all tools with consistent logging, timing, and tracing—without modifying each tool individually.

// Generic tool instrumentation wrapper
function instrumentTool<T>(
  toolName: string,
  fn: (...args: any[]) => Promise<any>
): typeof fn {
  return async (...args) => {
    const startTime = performance.now();
    const span = tracer.startSpan(`tool.${toolName}`);
    
    try {
      logger.info(`[${toolName}] Starting`, { args: sanitizeArgs(args) });
      
      const result = await fn(...args);
      const duration = performance.now() - startTime;
      
      metrics.recordToolLatency(toolName, duration);
      logger.info(`[${toolName}] Completed`, { duration: `${duration.toFixed(2)}ms` });
      span.setStatus({ code: SpanStatusCode.OK });
      
      return result;
    } catch (error) {
      span.setStatus({ code: SpanStatusCode.ERROR });
      span.recordException(error);
      metrics.incrementToolErrors(toolName);
      throw error;
    } finally {
      span.end();
    }
  };
}

// Apply to all tools automatically at initialization
function instrumentTools(tools: Record<string, CoreTool>): Record<string, CoreTool> {
  return Object.fromEntries(
    Object.entries(tools).map(([name, tool]) => [
      name,
      { ...tool, execute: instrumentTool(name, tool.execute) }
    ])
  );
}

// Usage: wrap once, observe everything
const tools = instrumentTools({ shell, filesystem, memory, ... });

Deep Dive Evaluating RAG Quality

Recall@k MRR (Mean Reciprocal Rank) NDCG Context Sufficiency 
// Recall@k: What % of relevant docs are in top-k?
function recallAtK(
  retrieved: string[],
  relevant: string[],
  k: number
): number {
  const topK = new Set(retrieved.slice(0, k));
  const found = relevant.filter(doc => topK.has(doc)).length;
  return found / relevant.length;
}

// MRR: Average of 1/rank of first relevant result
function mrr(queries: QueryResult[]): number {
  const reciprocalRanks = queries.map(q => {
    const rank = q.retrieved.findIndex(d => q.relevant.includes(d));
    return rank >= 0 ? 1 / (rank + 1) : 0;
  });
  return reciprocalRanks.reduce((a, b) => a + b) / queries.length;
}

// Evaluation Harness Structure
interface BenchmarkTask {
  id: string;
  prompt: string;
  expectedOutput: string | ((output: string) => boolean);
  category: 'code' | 'reasoning' | 'tool_use';
  difficulty: 'easy' | 'medium' | 'hard';
}

interface BenchmarkResult {
  taskId: string;
  passed: boolean;
  executionTime: number;
  tokenUsage: { input: number; output: number };
  toolCalls: number;
}

class EvaluationHarness {
  private tasks: BenchmarkTask[];
  private results: BenchmarkResult[] = [];
  
  async runBenchmark(agent: Agent): Promise<BenchmarkScore> {
    for (const task of this.tasks) {
      const start = Date.now();
      
      const result = await agent.run(task.prompt);
      
      const passed = typeof task.expectedOutput === 'function'
        ? task.expectedOutput(result.output)
        : result.output.includes(task.expectedOutput);
      
      this.results.push({
        taskId: task.id,
        passed,
        executionTime: Date.now() - start,
        tokenUsage: result.usage,
        toolCalls: result.toolCalls.length,
      });
    }
    
    return this.calculateScore();
  }
  
  private calculateScore(): BenchmarkScore {
    const passed = this.results.filter(r => r.passed).length;
    return {
      passRate: passed / this.results.length,
      avgLatency: avg(this.results.map(r => r.executionTime)),
      byDifficulty: this.groupByDifficulty(),
    };
  }
}

// 1. Connection pooling for multiple servers
class MCPManager {
  private clients = new Map<string, MCPClient>();
  
  async getClient(serverName: string): Promise<MCPClient> {
    if (!this.clients.has(serverName)) {
      const client = await this.connect(serverName);
      this.clients.set(serverName, client);
    }
    return this.clients.get(serverName)!;
  }
  
  async closeAll(): Promise<void> {
    await Promise.all(
      [...this.clients.values()].map(c => c.close())
    );
  }
}

  Object.entries(mcpTools).map(([name, tool]) => 
    [`mcp_${serverName}_${name}`, tool]
  )
);
